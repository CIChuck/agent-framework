checks = []

def command(args, *, cwd=W, input=None):
    result = subprocess.run(args, cwd=cwd, env=env, input=input,
                            text=True, capture_output=True, timeout=60)
    receipt = {'argv': args, 'cwd': str(cwd), 'exit_code': result.returncode,
               'stdout': result.stdout, 'stderr': result.stderr}
    checks.append(receipt)
    (B / 'independent-command-receipts.json').write_text(json.dumps(checks, indent=2) + '\n')
    print(args[-1], result.returncode, flush=True)
    if result.returncode:
        raise RuntimeError(receipt)
    return result.stdout

hashes = {path: hashlib.sha256((W / path).read_bytes()).hexdigest()
          for path in request['authority_sha256']}
assert hashes == request['authority_sha256'], hashes

comparison = []
source_files = {
    'friday-chat': ['config.py', 'runtime.py', 'display.py', 'cli.py', 'adapters.py', 'background.py', 'app.py', 'reorganization/assembly.py'],
    'friday-llm': ['model_library.py'],
    'friday-runtime': ['model_library.py'],
}
for package, files in source_files.items():
    module = package.replace('-', '_')
    dist_name = 'friday-chat' if package == 'friday-chat' else package.replace('friday-', 'friday-framework-')
    dist = metadata.distribution(dist_name)
    for rel in files:
        source = W / 'packages' / package / 'src' / module / rel
        wheel = Path(dist.locate_file(module + '/' + rel))
        source_hash = hashlib.sha256(source.read_bytes()).hexdigest()
        wheel_hash = hashlib.sha256(wheel.read_bytes()).hexdigest()
        comparison.append({'source': str(source.relative_to(W)), 'installed': str(wheel), 'source_sha256': source_hash, 'installed_sha256': wheel_hash, 'equal': source_hash == wheel_hash})
assert all(item['equal'] for item in comparison), comparison
