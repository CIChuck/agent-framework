def req_key(reqtext):
 req=Requirement(reqtext)

def expected_requirements(project):
 result={req_key(r) for r in project.get('dependencies',[])}
 extras=project.get('optional-dependencies',{})
 def expand(extra,seen=()):
  require(extra not in seen,'Recursive self extra: '+project['name'])
  if extra in seen:return []
  output=[]
  for text in extras.get(extra,[]):
   req=Requirement(text)
   if canonicalize_name(req.name)==canonicalize_name(project['name']):
    for child in req.extras: output.extend(expand(child,seen+(extra,)))
   else: output.append(text)
  return output
 for extra in extras:
  for text in expand(extra):
   req=Requirement(text)
   marker='extra == '+repr(canonicalize_name(extra))
   if req.marker: marker='('+str(req.marker)+') and '+marker
   bare=canonicalize_name(req.name)
   if req.extras: bare+='['+','.join(sorted(canonicalize_name(e) for e in req.extras))+']'
   bare+=str(req.specifier)
   result.add(req_key(bare+'; '+marker))
 return result
