def slug(s):
    s=re.sub(r'<[^>]+>','',s).replace('`','').lower()
    s=''.join(c for c in s if c in ' -_' or unicodedata.category(c)[0] in 'LN')
    return s.replace(' ','-')

def anchors(p):
    s=p.read_text(); fence=False; found=set(); counts={}
    for l in s.splitlines():
        if l.startswith(('```','~~~')):fence=not fence
        if not fence and re.match(r'^#{1,6} ',l):
            key=slug(l.split(' ',1)[1]);n=counts.get(key,0);counts[key]=n+1
            found.add(key if n==0 else key+'-'+str(n))
    found.update(re.findall(r'<a\s+(?:id|name)=["\']([^"\']+)',s))
    return found
