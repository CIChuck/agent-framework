import ast, hashlib, importlib.metadata as md, json, pathlib, re, subprocess, tomllib
W=pathlib.Path('/private/tmp/friday-guides-refresh');O=pathlib.Path('/private/tmp/friday-guides-final-independent-review')
packages=[]; source_comparison=[]; inventory=[]
for pyproject in sorted((W/'packages').glob('*/pyproject.toml')):
    t=tomllib.loads(pyproject.read_text());p=t['project'];dist=md.distribution(p['name']);mismatches=[];count=0
    for source in (pyproject.parent/'src').rglob('*'):
        if not source.is_file() or '__pycache__' in source.parts or source.name == '.DS_Store':continue
        rel=source.relative_to(pyproject.parent/'src');installed=pathlib.Path(dist.locate_file(rel));count+=1
        if not installed.is_file() or source.read_bytes()!=installed.read_bytes():mismatches.append(str(rel))
    packages.append({'directory':pyproject.parent.name,'distribution':p['name'],'version':p['version'],'installed_version':dist.version,'python':p['requires-python'],'extras':sorted(p.get('optional-dependencies',{})),'wheel_packages':t.get('tool',{}).get('hatch',{}).get('build',{}).get('targets',{}).get('wheel',{}).get('packages',[])})
    source_comparison.append({'package':p['name'],'files_compared':count,'mismatches':mismatches})
for source in sorted((W/'packages/friday-tools/src').rglob('*.py')):
    tree=ast.parse(source.read_text())
    for node in tree.body:
        if not isinstance(node,ast.ClassDef):continue
        name=None
        for sub in node.body:
            if isinstance(sub,(ast.Assign,ast.AnnAssign)):
                targets=sub.targets if isinstance(sub,ast.Assign) else [sub.target]
                if any(isinstance(target,ast.Name) and target.id=='name' for target in targets):
                    if isinstance(sub.value,ast.Constant) and isinstance(sub.value.value,str): name=sub.value.value
            if isinstance(sub,ast.FunctionDef) and sub.name=='name':
                for v in ast.walk(sub):
                    if isinstance(v,ast.Return) and isinstance(v.value,ast.Constant) and isinstance(v.value.value,str):name=v.value.value
        if name:
            constants=[v.value for v in ast.walk(node) if isinstance(v,ast.Constant) and isinstance(v.value,str)]
            stub=any(re.search(r'not (?:yet )?implemented|\bstub\b',v,re.I) for v in constants)
            inventory.append({'class':node.name,'name':name,'path':str(source.relative_to(W)),'line':node.lineno,'stub':stub})
rows=[]
for lineno,line in enumerate((W/'docs/guides/14-appendices.md').read_text().splitlines(),1):
    if not line.startswith('| '):continue
    match=re.search(r'\| `([^`]+)` \| \[([^]]+)\]\(([^)]+)\).*\| (stub|implementation present) \|',line)
    if match:
        name,cls,path,status=match.groups();rows.append({'name':name,'class':cls,'status':status,'line':lineno})
by_class={r['class']:r for r in rows}
missing=[x for x in inventory if x['class'] not in by_class]
stub_mismatch=[{'source':x,'document':by_class[x['class']]} for x in inventory if x['class'] in by_class and x['stub']!=(by_class[x['class']]['status']=='stub')]
base='123d9ed943519c4e68bf9249705e55d1a1ebf8b8';candidate='6fb53e00877214d813656b204f950c8a0b725d05'
diff=subprocess.check_output(['git','diff','--unified=0',base+'..'+candidate,'--','docs/guides'],cwd=W,text=True)
(O/'guide-diff.patch').write_text(diff)
structure=[l for l in diff.splitlines() if l.startswith(('diff --git','@@','+#','-#'))]
(O/'guide-diff-structure.txt').write_text('\n'.join(structure)+'\n')
result={'packages':packages,'source_comparison':source_comparison,'tool_inventory':inventory,'catalog_rows':len(rows),'catalog_missing':missing,'stub_status_mismatches':stub_mismatch}
(O/'static-review.json').write_text(json.dumps(result,indent=2)+'\n')
print('packages',len(packages),'package_version_matches',all(p['version']==p['installed_version']=='0.1.0a3' for p in packages))
print('package_files_compared',sum(p['files_compared'] for p in source_comparison),'package_mismatches',[(p['package'],p['mismatches']) for p in source_comparison if p['mismatches']])
print('catalog_rows',len(rows),'statically_named_classes',len(inventory),'missing',missing)
print('stub_status_mismatches',json.dumps(stub_mismatch))
