import os,sys
PRIVATE='/private/tmp/friday-chat-quick-start-validation/'
ROOT='/Users/chuckrussell/Dev/gendev/agent-framework'
def guard(event,args):
 if event in ('socket.getaddrinfo','socket.connect','socket.sendto'):raise RuntimeError('Network prohibited')
 if event=='open' and isinstance(args[0],(str,bytes,os.PathLike)):
  path=os.path.abspath(os.fsdecode(args[0]))
  if path==ROOT or path.startswith(ROOT+'/'):raise RuntimeError('Shared checkout access prohibited')
  if '.env' in path.split('/') or path.endswith('/scripts/pypi.sh'):raise RuntimeError('Credential file access prohibited')
  flags=args[2] if len(args)>2 and isinstance(args[2],int) else 0
  mode=args[1] if len(args)>1 and isinstance(args[1],str) else ''
  if (flags & (os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND) or any(v in mode for v in 'wax+')) and not path.startswith(PRIVATE) and path!='/dev/null':raise RuntimeError('Write outside private validation '+path)
sys.addaudithook(guard)
